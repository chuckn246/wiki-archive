====Script====
<code>
#!/bin/sh

PREREQ=""
prereqs()
{
     echo "$PREREQ"
}

case $1 in
prereqs)
     prereqs
     exit 0
     ;;
esac

# Solarized Light
echo "253,220,133,181,38,211,42,7,238,203,147,131,101,108,88,0
246,50,153,137,139,54,161,54,232,75,161,148,123,113,110,43
227,47,0,0,210,130,152,66,213,22,161,150,131,196,117,54" | setvtrgb -
</code>
