====Script====

<code>
#!/bin/sh

PREREQ=""
prereqs()
{
     echo "$PREREQ"
}

case $1 in
prereqs)
     prereqs
     exit 0
     ;;
esac

# Gruvbox Light
echo "251,204,152,215,69,177,104,124,146,157,121,181,7,143,66,60
241,36,151,153,133,98,157,111,131,0,116,118,102,63,123,56
199,29,26,33,136,134,106,100,116,6,14,20,120,113,88,54" | setvtrgb -
</code>
