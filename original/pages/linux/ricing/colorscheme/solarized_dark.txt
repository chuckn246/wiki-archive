====Scripts====

<file sh solarized_dark>
#!/bin/sh

PREREQ=""
prereqs()
{
     echo "$PREREQ"
}

case $1 in
prereqs)
     prereqs
     exit 0
     ;;
esac

# Solarized Dark
echo "0,220,133,181,38,211,42,238,7,203,88,101,131,108,147,253
43,50,153,137,139,54,161,232,54,75,110,123,148,113,161,246
