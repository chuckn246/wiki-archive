====Script====

<file sh dracula>
#!/bin/sh

PREREQ=""
prereqs()
{
     echo "$PREREQ"
}

case $1 in
prereqs)
     prereqs
     exit 0
     ;;
esac

# Dracula
echo "40,255,80,241,189,255,139,191,77,255,90,244,202,255,154,230
